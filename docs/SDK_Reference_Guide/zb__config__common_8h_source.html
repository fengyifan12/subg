346 /* ZB packet length must not exceed 127 bytes
347  *
348  * Old calculation by Maxim:
349  *
350  * MAC header (total: 11 bytes)
351  * - FCF: 2 bytes
352  * - Sequence number: 1 bytes
353  * - PAN: 2 bytes
354  * - Destination address: 2 bytes
355  * - Source address: 2 bytes
356  *
357  * NWK header (total: 24 bytes)
358  * - FCF: 2 bytes
359  * - Destination short address: 2 bytes
360  * - Source short address: 2 bytes
361  * - Radius: 1 byte
362  * - Sequence number: 1 byte
363  * - Destination long address: 8 bytes
364  * - Source long address: 8 bytes
365  *
366  * ZB security header (total: 31 bytes)
367  *
368  * APS header (total: 8 bytes)
369  * - FCF: 1 byte
370  * - Destination endpoint: 1 byte
371  * - Cluster ID: 2 bytes
372  * - Profile ID: 2 bytes
373  * - Source endpoint: 1 byte
374  * - Counter: 1 byte
375  *
376  * APS payload (total: variable length)
377  *
378  * MAC footer (total: 2 bytes)
379  * - FCS: 2 bytes
380  *
381  * Total length w/o APS payload: 127-(11+24+31+8+2)= 51 bytes,
382  * consequently APS payload max size is 51 bytes.
383  */
