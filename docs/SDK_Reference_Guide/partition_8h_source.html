168 __STATIC_INLINE void TZ_SAU_Setup (void)
169 {
170 
171  uint32_t flash_model_info, flash_size;
172  uint32_t chip_id = SYSCTRL->soc_chip_info.bit.chip_id;
173 
174 #if defined (SCB_CSR_AIRCR_INIT) && (SCB_CSR_AIRCR_INIT == 1U)
175  SCB->SCR = (SCB->SCR & ~(SCB_SCR_SLEEPDEEPS_Msk )) |
176  ((SCB_CSR_DEEPSLEEPS_VAL << SCB_SCR_SLEEPDEEPS_Pos) & SCB_SCR_SLEEPDEEPS_Msk);
177 
178  SCB->AIRCR = (SCB->AIRCR & ~(SCB_AIRCR_VECTKEY_Msk | SCB_AIRCR_SYSRESETREQS_Msk |
179  SCB_AIRCR_BFHFNMINS_Msk | SCB_AIRCR_PRIS_Msk )) |
180  ((0x05FAU << SCB_AIRCR_VECTKEY_Pos) & SCB_AIRCR_VECTKEY_Msk) |
181  ((SCB_AIRCR_SYSRESETREQS_VAL << SCB_AIRCR_SYSRESETREQS_Pos) & SCB_AIRCR_SYSRESETREQS_Msk) |
182  ((SCB_AIRCR_PRIS_VAL << SCB_AIRCR_PRIS_Pos) & SCB_AIRCR_PRIS_Msk) |
183  ((SCB_AIRCR_BFHFNMINS_VAL << SCB_AIRCR_BFHFNMINS_Pos) & SCB_AIRCR_BFHFNMINS_Msk);
184 #endif /* defined (SCB_CSR_AIRCR_INIT) && (SCB_CSR_AIRCR_INIT == 1U) */
185 
186 #if defined (__FPU_USED) && (__FPU_USED == 1U) && \
187  defined (TZ_FPU_NS_USAGE) && (TZ_FPU_NS_USAGE == 1U)
188 
189  SCB->NSACR = (SCB->NSACR & ~(SCB_NSACR_CP10_Msk | SCB_NSACR_CP11_Msk)) |
190  ((SCB_NSACR_CP10_11_VAL << SCB_NSACR_CP10_Pos) & (SCB_NSACR_CP10_Msk | SCB_NSACR_CP11_Msk));
191 
192  FPU->FPCCR = (FPU->FPCCR & ~(FPU_FPCCR_TS_Msk | FPU_FPCCR_CLRONRETS_Msk | FPU_FPCCR_CLRONRET_Msk)) |
193  ((FPU_FPCCR_TS_VAL << FPU_FPCCR_TS_Pos ) & FPU_FPCCR_TS_Msk ) |
194  ((FPU_FPCCR_CLRONRETS_VAL << FPU_FPCCR_CLRONRETS_Pos) & FPU_FPCCR_CLRONRETS_Msk) |
195  ((FPU_FPCCR_CLRONRET_VAL << FPU_FPCCR_CLRONRET_Pos ) & FPU_FPCCR_CLRONRET_Msk );
196 #endif
197 
198  TZ_SAU_Disable();
199 
200  SEC_CTRL->sec_peri_attr[0] = 0x0;
201  SEC_CTRL->sec_peri_attr[1] = 0x0;
202  SEC_CTRL->sec_peri_attr[2] = 0x0;
203 
204  flash_model_info = inp32(FLASH_SIZE_INFO);
205  flash_size = (1 << ((flash_model_info >> 16) & 0xFF));
206  flash_size = (flash_size >> 5); /*32 bytes uints*/
207 
208  /*set all flash in secure mode*/
209  if(chip_id==0x0584)
210  {
211  SEC_CTRL->sec_flash_sec_size = 0x8000;
212  SEC_CTRL->sec_flash_nsc_start = 0x8000;
213  SEC_CTRL->sec_flash_nsc_stop = 0x8000;
214  SEC_CTRL->sec_flash_ns_stop = flash_size;
215  }
216  else
217  {
218  SEC_CTRL->sec_flash_sec_size = 0x10000;
219  SEC_CTRL->sec_flash_nsc_start = 0x10000;
220  SEC_CTRL->sec_flash_nsc_stop = 0x10000;
221  SEC_CTRL->sec_flash_ns_stop = flash_size;
222  }
223 
224  /*
225  * Here assume 64KB secure... NSC in 62KB. 0x400 bytes.
226  * Sec address Non Sec address
227  * RAM0 0x30000000 0x30007FFF 32KB RAM0 0x20000000 0x20007FFF 32KB
228  * RAM1 0x30008000 0x3000FFFF 32KB RAM1 0x20008000 0x2000FFFF 32KB
229  * RAM2 0x30010000 0x30017FFF 32KB RAM2 0x20010000 0x20017FFF 32KB
230  * RAM3 0x30018000 0x3001FFFF 32KB RAM3 0x20018000 0x2001FFFF 32KB
231  * RAM4 0x30020000 0x30027FFF 32KB RAM4 0x20020000 0x20027FFF 32KB
232  * RAM5 0x30028000 0x3002BFFF 16KB RAM5 0x20028000 0x2002BFFF 16KB
233  * RAM6 0x3002C000 0x3002FFFF 16KB RAM6 0x2002C000 0x2002FFFF 16KB
234  *
235  * set the sram to be secure during bootrom time
236  * For bootrom we set all SRAM (192KB as secure).
237  * NO NSC in SRAM. so SEC_RAM_NSC_START = SEC_RAM_NSC_STOP
238  */
239  SEC_CTRL->sec_ram_sec_size = 0x1800;
240  SEC_CTRL->sec_ram_nsc_start = 0x1800;
