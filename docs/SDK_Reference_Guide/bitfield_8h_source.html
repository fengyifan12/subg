137 {
138  /* Lookup table for offset into a single bitfield block. */
139  static const uint8_t multiply_de_bruijn_bit_position[32] =
140  {
141  0, 1, 28, 2, 29, 14, 24, 3, 30, 22, 20, 15, 25, 17, 4, 8,
142  31, 27, 13, 23, 21, 19, 16, 7, 26, 12, 18, 6, 11, 5, 10, 9
143  };

145  if (start >= bits)
146  {
147  return bits;
148  }
149

151  uint32_t v = p_bitfield[i] & ~(BITFIELD_MASK(start) - 1);
152  while (v == 0)
