68 /* uECC_RNG_Function type
69 The RNG function should fill 'size' random bytes into 'dest'. It should return 1 if
70 'dest' was filled with random data, or 0 if the random data could not be generated.
71 The filled-in values should be either truly random, or from a cryptographically-secure PRNG.
72 
73 A correctly functioning RNG function must be set (using uECC_set_rng()) before calling
74 uECC_make_key() or uECC_sign().

77 for uECC_shared_secret() and uECC_sign_deterministic().

149 /* uECC_HashContext structure.
150 This is used to pass in an arbitrary hash function to uECC_sign_deterministic().
151 The structure will be used for multiple hash computations; each time a new hash
152 is computed, init_hash() will be called, followed by one or more calls to
153 update_hash(), and finally a call to finish_hash() to prudoce the resulting hash.
154 
155 The intention is that you will create a structure that includes uECC_HashContext
156 followed by any hash-specific data. For example:
157 
158 typedef struct SHA256_HashContext {
159  uECC_HashContext uECC;
160  SHA256_CTX ctx;
161 } SHA256_HashContext;
162 
163 void init_SHA256(uECC_HashContext *base) {
164  SHA256_HashContext *context = (SHA256_HashContext *)base;
165  SHA256_Init(&context->ctx);
166 }
167 
168 void update_SHA256(uECC_HashContext *base,
169  const uint8_t *message,
170  unsigned message_size) {
171  SHA256_HashContext *context = (SHA256_HashContext *)base;
172  SHA256_Update(&context->ctx, message, message_size);
173 }
174 
175 void finish_SHA256(uECC_HashContext *base, uint8_t *hash_result) {
176  SHA256_HashContext *context = (SHA256_HashContext *)base;
177  SHA256_Final(hash_result, &context->ctx);
178 }
179 
180 ... when signing ...
181 {
182  uint8_t tmp[32 + 32 + 64];
183  SHA256_HashContext ctx = {{&init_SHA256, &update_SHA256, &finish_SHA256, 64, 32, tmp}};
184  uECC_sign_deterministic(key, message_hash, &ctx.uECC, signature);
185 }
186 */
187 typedef struct uECC_HashContext {
188  void (*init_hash)(struct uECC_HashContext *context);
189  void (*update_hash)(struct uECC_HashContext *context,
190  const uint8_t *message,
191  unsigned message_size);
192  void (*finish_hash)(struct uECC_HashContext *context, uint8_t *hash_result);
193  unsigned block_size; /* Hash function block size in bytes, eg 64 for SHA-256. */
194  unsigned result_size; /* Hash function result size in bytes, eg 32 for SHA-256. */
195  uint8_t *tmp; /* Must point to a buffer of at least (2 * result_size + block_size) bytes. */
196 } uECC_HashContext;
