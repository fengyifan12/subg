133 #define ZB_ZCL_TUNNEL_SEND_TRANSFER_REQ( \
134  _buffer, _addr, _dst_addr_mode, _dst_ep, _ep, _prfl_id, _def_resp, _cb, \
135  _manufacturer_id, _flag, _byte_num, _data_size, _image_data) \
136 { \
137  zb_uint8_t* __ptr = ZB_ZCL_START_PACKET((_buffer)); \
138  ZB_ZCL_CONSTRUCT_SPECIFIC_COMMAND_REQ_FRAME_CONTROL_A(__ptr, \
139  ZB_ZCL_FRAME_DIRECTION_TO_SRV, ZB_ZCL_MANUFACTURER_SPECIFIC, (_def_resp));\
140  ZB_ZCL_CONSTRUCT_COMMAND_HEADER_EXT(__ptr, ZB_ZCL_GET_SEQ_NUM(), \
141  ZB_ZCL_MANUFACTURER_SPECIFIC, (_manufacturer_id), \
142  ZB_ZCL_CMD_TUNNEL_TRANSFER_DATA_REQ); \
143  ZB_ZCL_PACKET_PUT_DATA8(__ptr, (_flag)); \
144  ZB_ZCL_PACKET_PUT_DATA16_VAL(__ptr, (_byte_num)); \
145  ZB_ZCL_PACKET_PUT_DATA8(__ptr, (_data_size)); \
146  if((_data_size)>0) \
147  { \
148  ZB_ZCL_PACKET_PUT_DATA_N(__ptr, (_image_data), (_data_size)); \
149  } \
150  ZB_ZCL_FINISH_PACKET((_buffer), __ptr) \
151  ZB_ZCL_SEND_COMMAND_SHORT( \
152  (_buffer), (_addr), (_dst_addr_mode), (_dst_ep), (_ep), (_prfl_id), \
153  ZB_ZCL_CLUSTER_ID_TUNNEL, (_cb)); \
154 }

165 #define ZB_ZCL_TUNNEL_GET_TRANSFER_REQ(_data_ptr, _buffer, _status) \
166 { \
167  zb_zcl_tunnel_transfer_data_req_t *src_ptr = \
168  (zb_zcl_tunnel_transfer_data_req_t*)zb_buf_begin((_buffer)); \
169  \
170  if (zb_buf_len((_buffer)) != sizeof(zb_zcl_tunnel_transfer_data_req_t) \
171  + src_ptr->data_size * sizeof(zb_uint8_t) ) \
172  { \
173  (_status) = ZB_ZCL_PARSE_STATUS_FAILURE; \
174  } \
175  else \
176  { \
177  (_status) = ZB_ZCL_PARSE_STATUS_SUCCESS; \
178  (_data_ptr)->req_header.tx_flag = src_ptr->tx_flag; \
179  ZB_HTOLE16((zb_uint8_t*)&((_data_ptr)->req_header.byte_num), (zb_uint8_t*)&(src_ptr->byte_num)); \
180  (_data_ptr)->req_header.data_size = src_ptr->data_size; \
181  (_data_ptr)->tun_data = (zb_uint8_t*)(src_ptr+1); \
182  } \
183 }
