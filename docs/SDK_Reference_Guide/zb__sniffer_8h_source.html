37 /*
38  * Page and channel may be received as 2 bytes or packed in a single byte.
39  * For single byte cases, the following pattern is found:
40  * -> 0bPPPC CCCC (P - Page, C - Channel).
41  * -> 3 MS bits for pages 0, [28;31].
42  * -> remaining 5 bits for channel.
43  *
44  * Out of the 8 combinations for the 'single byte' interval, 3 are free and we use 1 to flag that it is a 2 byte scheme.
45  * 0b000c cccc -> page 0
46  * 0b001c cccc -> page 28
47  * 0b010c cccc -> page 29
48  * 0b011c cccc -> page 30
49  * 0b100c cccc -> page 31
50  * 0b101c cccc -> free
51  * 0b110c cccc -> free
52  * 0b111c cccc -> used by sniffer GUI/console to indicate 2 byte scenario
53  */
54 
55 /*
56  * In single byte schemes, used in legacy devices, the following definitions
57  * allow to apply the offset from page 0 up to page 31 within the
58  * {0, [28;31]} interval.
59  */
60 #define ZB_PAGE_AND_CHANNEL_PACKED_OFFSET 5U
61 #define ZB_CHANNEL_LIST_PAGE_0_PACKED_IDX ZB_CHANNEL_LIST_PAGE0_IDX
62 #define ZB_CHANNEL_LIST_PAGE_28_PACKED_IDX ZB_CHANNEL_LIST_PAGE28_IDX - ZB_PAGE_AND_CHANNEL_PACKED_OFFSET // 1
63 #define ZB_CHANNEL_LIST_PAGE_29_PACKED_IDX ZB_CHANNEL_LIST_PAGE29_IDX - ZB_PAGE_AND_CHANNEL_PACKED_OFFSET // 2
64 #define ZB_CHANNEL_LIST_PAGE_30_PACKED_IDX ZB_CHANNEL_LIST_PAGE30_IDX - ZB_PAGE_AND_CHANNEL_PACKED_OFFSET // 3
65 #define ZB_CHANNEL_LIST_PAGE_31_PACKED_IDX ZB_CHANNEL_LIST_PAGE31_IDX - ZB_PAGE_AND_CHANNEL_PACKED_OFFSET // 4
