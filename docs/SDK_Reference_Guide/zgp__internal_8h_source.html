320 /* ZGP spec, A.3.6.1.3.1:
321  *
322  * For the incremental sequence number (when SecurityLevel = 0b00), the
323  * counter must be allowed to roll over, because of the limited
324  * sequence number length of 1 octet, so care must be taken when
325  * comparing for freshness. It is recommended that this comparison be
326  * accomplished as follows:
327  * define
328  * a = sequence number stored by the GPP/GPS;
329  * b = sequence number from the GPDF;
330  * if 1 ≤ ( (b – a)mod 256 ) < 128 accept GPDF;
331  * else drop GPDF;
332  */

334 /* NK: Skip freshness check for some special cases, for example, to do not drop packets after ZGP
335  * reset. This behaviour is not clear in general, seems like it is temporary feature. */
336 #ifdef ZB_ZGP_SKIP_FRESHNESS_CHECK
337 #define ZB_ZGP_CHECK_MAC_SEQ_NUM_FRESHNESS(prev_sn, new_sn, ret) \
338 { \
339  ret = ZB_TRUE; \
340 }
341 #else
342 #define ZB_ZGP_CHECK_MAC_SEQ_NUM_FRESHNESS(prev_sn, new_sn, ret) \
343 { \
344  /* Modular arithmetic takes place here, result value is "wrap around" */ \
345  /* NK: Another overflow processing. */ \
346  /* For example, prev_sn is 125, new_sn is 9; in this case: */ \
347  /* (new_sn - prev_sn) = 9 - 125 (overflowed) = 255 - 116 = 139 > 128 - frame dropped */ \
348  /* abs(new_sn - prev_sn) = 125 - 9 = 116 < 128 - frame passed */ \
349  /* It is needed to avoid too many drop situations after ZGP reset. */ \
350  /* NK:TODO: We do not store this counter in nvram, so when GW reset occurs, it will skip some pkts. */ \
351  zb_uint8_t diff = (new_sn > prev_sn) ? (new_sn - prev_sn) : (prev_sn - new_sn); \
352  /* Check that diff in seq nums falls into half of seq num value range */ \
353  ret = ((diff > 0) && (diff < 128)) ? ZB_TRUE : ZB_FALSE; \
