280 __STATIC_INLINE uint32_t gpio_pin_toggle(uint32_t pin_number) {
281  uint32_t state, MASK;
282 
283  if( pin_number >= MAX_NUMBER_OF_PINS ) {
284  return STATUS_INVALID_PARAM;
285  }
286 
287  MASK = (1 << pin_number);
288  state = GPIO->output_state & MASK;
289 
290  if (state) {
291  GPIO->output_low = MASK;
292  } else {
293  GPIO->output_high = MASK;
294  }
295  return STATUS_SUCCESS;
296 }
