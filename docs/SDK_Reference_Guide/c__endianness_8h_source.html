105 # define htobe32(x) (x)

107 # define htole32(x) \

109  uint32_t _temp = (x); \
110  ((_temp >> 24) & 0x000000FF) | ((_temp >> 8) & 0x0000FF00) | \
111  ((_temp << 8) & 0x00FF0000) | ((_temp << 24) & 0xFF000000); \
112  }))
