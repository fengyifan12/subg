38  /* Common logic when we have problems with parent polling:
39  * 1) Data Request + 3 MAC Retries
40  * 2) Wait MACPollFailureWaitTime
41  * 3) Repeat steps 1-2 wwah_retry_cnt times
42  * 4) Wait work_poll_interval
43  * 5) Repeat steps 1-4 ZDO_CTX().max_parent_threshold_retry times
44  */
45 
46  /* Note:
47  * 1) if current retry count(crc) equals zero it means that last poll was success
48  * 2) crc >= maximum_retry_count leads to polling stop
49  * 3) crc % (wwah_retry_cnt + 1) shows current iteration of additional polling,
50  * i.e. loop steps 1-2 above. If remainder is zero - wait for standart timeout,
51  * or additional timeout otherwise.
52  */
53 
54 #define ZB_ZDO_GET_CURENT_POLL_ITERATION_INTERVAL(wait_time) \
55  if (zb_zcl_wwah_mac_poll_cca_retry_count() && \
56  ZDO_CTX().parent_threshold_retry % (1 + zb_zcl_wwah_mac_poll_cca_retry_count())) \
57  { \
58  (wait_time) = zb_zcl_wwah_mac_poll_failure_wait_time(); \
59  } \
60  else \
61  { \
62  (wait_time) = ZDO_CTX().pim.work_poll_interval; \
63  }
