255  \
256  pair_req_ptr = (zb_buf_len(data_buf) >= req_size) ? \

265  { \

304 #define ZB_ZCL_THERMOSTAT_INIT_SET_WEEKLY_SCHEDULE_REQ(buffer, \
305  cmd_ptr, def_resp, num_of_transition, day_of_week, mode_for_seq, \
306  transition, heat_set, cool_set) \
307 { \
308  cmd_ptr = ZB_ZCL_START_PACKET_REQ(buffer) \
309  ZB_ZCL_CONSTRUCT_SPECIFIC_COMMAND_REQ_FRAME_CONTROL(cmd_ptr, def_resp) \
310  ZB_ZCL_CONSTRUCT_COMMAND_HEADER_REQ(cmd_ptr, ZB_ZCL_GET_SEQ_NUM(), \
311  ZB_ZCL_CMD_THERMOSTAT_SET_WEEKLY_SCHEDULE); \
312  ZB_ZCL_PACKET_PUT_DATA8(cmd_ptr, (num_of_transition)); \
313  ZB_ZCL_PACKET_PUT_DATA8(cmd_ptr, (day_of_week)); \
314  ZB_ZCL_PACKET_PUT_DATA8(cmd_ptr, (mode_for_seq)); \
315  ZB_ZCL_PACKET_PUT_DATA16_VAL(cmd_ptr, (transition)); \
316  if (ZB_ZCL_THERMOSTAT_WEEKLY_SCHEDULE_MODE_FOR_SEQ_HEAT == (mode_for_seq)) \
317  { \
318  ZB_ZCL_PACKET_PUT_DATA16_VAL(cmd_ptr, (heat_set)); \
319  } \
320  else \
321  if (ZB_ZCL_THERMOSTAT_WEEKLY_SCHEDULE_MODE_FOR_SEQ_COOL == (mode_for_seq)) \
322  { \
323  ZB_ZCL_PACKET_PUT_DATA16_VAL(cmd_ptr, (cool_set)); \
324  } \
325  else \
326  if (ZB_ZCL_THERMOSTAT_WEEKLY_SCHEDULE_MODE_FOR_SEQ_BOTH == (mode_for_seq)) \
327  { \
